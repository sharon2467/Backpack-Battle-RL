def raw(path):
    p = Path(path)
    if not p.is_absolute(): p = R/p
    for q in reversed((p,) + tuple(p.parents)):
        s = q.lstat()
        assert not stat.S_ISLNK(s.st_mode)
        assert stat.S_ISDIR(s.st_mode) if q != p else stat.S_ISREG(s.st_mode) and s.st_nlink == 1
    return p.read_bytes()

def desc(path):
    b = raw(path)
    return {'path':str(path),'size':len(b),'sha256':hashlib.sha256(b).hexdigest()}

def load(path): return json.loads(raw(path))
def save(path, value):
    b = value.encode() if isinstance(value,str) else (json.dumps(value,ensure_ascii=False,indent=2)+'\n').encode()
    p = Path(path)
    if not p.is_absolute(): p = R/p
    with p.open('xb') as f: f.write(b)
    assert raw(p) == b
    return desc(str(p.relative_to(R)))

def version(s):
    for a,b in VERSION: s = s.replace(a,b)
    return s

def segment(s, name):
    n = next(n for n in ast.parse(s).body if isinstance(n,(ast.FunctionDef,ast.ClassDef)) and n.name == name)
    return ''.join(s.splitlines(keepends=True)[n.lineno-1:n.end_lineno])

def replace_segment(s,name,replacement):
    old = segment(s,name)
    assert s.count(old) == 1
    return s.replace(old,replacement,1)
