def identity(info):
    return [info.st_dev,info.st_ino,info.st_mode,info.st_nlink,info.st_size,
            info.st_mtime_ns,info.st_ctime_ns]


def read(pin):
    name = pin['path']
    relative = PurePosixPath(name)
    if not relative.is_absolute():
        assert relative.as_posix() == name and all(part not in ('','.','..') for part in name.split('/'))
    path = Path(name) if relative.is_absolute() else ROOT.joinpath(*relative.parts)
    assert path.resolve(strict=True) == path
    assert not any(p.is_symlink() for p in (path,*path.parents))
    before = path.lstat()
    assert stat.S_ISREG(before.st_mode) and before.st_nlink == 1 and before.st_size == pin['size']
    fd = os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        assert identity(os.fstat(fd)) == identity(before)
        chunks=[];remaining=pin['size']+1
        while remaining:
            chunk=os.read(fd,remaining)
            if chunk == b'':break
            chunks.append(chunk);remaining-=len(chunk)
        raw=b''.join(chunks)
        assert identity(os.fstat(fd)) == identity(before)
    finally:
        os.close(fd)
    assert identity(path.lstat()) == identity(before)
    assert len(raw) == pin['size'] and hashlib.sha256(raw).hexdigest() == pin['sha256'],name
    readback.append(dict(**pin,identity=identity(before)))
    raws[name]=raw
    return raw


def exact(left,right):
    return json.dumps(left,sort_keys=True,separators=(',',':')) == json.dumps(right,sort_keys=True,separators=(',',':'))


def key(row):

def resolve(pointer,index):
    assert type(pointer) is str and pointer.startswith('/') and pointer[1:].isdigit()
    number=int(pointer[1:]);assert pointer=='/'+str(number)
    return index[number]
