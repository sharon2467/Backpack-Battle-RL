def raw(path):
    relative = Path(path)
    assert not relative.is_absolute() and '..' not in relative.parts
    leaf = R / relative
    assert not any(p.is_symlink() for p in (leaf, *leaf.parents))
    info = leaf.lstat()
    assert stat.S_ISREG(info.st_mode) and info.st_nlink == 1
    value = leaf.read_bytes()
    after = leaf.lstat()
    assert (info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns, info.st_ctime_ns) == (after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns, after.st_ctime_ns)
    return value


def descriptor(path):
    value = raw(path)
    return dict(path=path, size=len(value), sha256=hashlib.sha256(value).hexdigest())


def checked(row):
    assert descriptor(row['path']) == row, row['path']
    return raw(row['path'])


def save(path, value):
    text = json.dumps(value, indent=2, sort_keys=True, ensure_ascii=False, allow_nan=False) + '\n'
    write(path, text)
    return descriptor(path)


def write(path, text):
    assert path.startswith(PREFIX) or path in {'tests/phase434_king_observer_schema.py', 'tests/phase434_oracle_host.py', 'tests/phase434_oracle_storage.py'}
    leaf = R / path
    leaf.parent.mkdir(parents=True, exist_ok=True)
    with leaf.open('x', encoding='utf-8', newline='') as stream:
        stream.write(text)
    assert raw(path) == text.encode('utf-8')


def replace(text, before, after, count=1):
    assert text.count(before) == count, (before[:120], text.count(before), count)
    return text.replace(before, after)


def functions(text, python):
    if python:
        lines = text.splitlines(True)
        return {n.name: (ast.dump(n, include_attributes=False), ''.join(lines[n.lineno - 1:n.end_lineno]))
                for n in ast.parse(text).body if isinstance(n, (ast.FunctionDef, ast.ClassDef))}
    matches = list(re.finditer(r'^func (\w+)\([^\n]*\)(?:\s*->[^:\n]+)?\s*:\n(?:[\t ][^\n]*\n|\n)*', text, re.M))
    assert len(matches) == len(re.findall(r'^func ', text, re.M))
    return {m.group(1): (m.group(0).rstrip(), m.group(0).rstrip()) for m in matches}


def successor(before_path, after_path, text, changed=(), added=()):
    before = raw(before_path).decode('utf-8')
    assert before.encode('utf-8') == raw(before_path)
    python = before_path.endswith('.py')
    code = python or before_path.endswith('.gd')
    if code:
        old_funcs, new_funcs = functions(before, python), functions(text, python)
        assert not set(old_funcs) - set(new_funcs)
        actual_changes = sorted(n for n in old_funcs if old_funcs[n] != new_funcs[n])
        assert actual_changes == sorted(changed), (after_path, actual_changes, changed)
        assert sorted(set(new_funcs) - set(old_funcs)) == sorted(added)
        conserved = sorted(set(old_funcs) - set(changed))
        if python:
            compile(text, after_path, 'exec')  # compile only, no import/exec
    else:
        actual_changes, conserved = [], []
    old_lines, new_lines = before.splitlines(True), text.splitlines(True)
    offsets = [0]
    for line in new_lines:
        offsets.append(offsets[-1] + len(line))
    edits = []
    for tag, a, b, c, d in difflib.SequenceMatcher(None, old_lines, new_lines, autojunk=False).get_opcodes():
        if tag != 'equal':
            edits.append(dict(new_text_offset=offsets[c], old=''.join(old_lines[a:b]), new=''.join(new_lines[c:d])))
    inverse = text
    for edit in reversed(edits):
        i = edit['new_text_offset']
        assert inverse[i:i + len(edit['new'])] == edit['new']
        inverse = inverse[:i] + edit['old'] + inverse[i + len(edit['new']):]
    assert inverse.encode('utf-8') == raw(before_path)
    write(after_path, text)
    stem = Path(after_path).stem
    record = dict(before=descriptor(before_path), after=descriptor(after_path),
                  changed_functions=actual_changes, added_functions=list(added), conserved_functions=conserved,
                  inverse='EXACT_COMPLETE_UTF8_BYTES', offset_unit='unicode-text-index; final byte equality checked', edits=edits,
                  python_compile_only=python, project_executions=0)
    save(PREFIX + stem + '-inverse.json', record)
    write(PREFIX + stem + '-source.patch', ''.join(difflib.unified_diff(old_lines, new_lines, fromfile=before_path, tofile=after_path)))
    CONSERVATION.append({k: v for k, v in record.items() if k != 'edits'})
    return descriptor(after_path)
