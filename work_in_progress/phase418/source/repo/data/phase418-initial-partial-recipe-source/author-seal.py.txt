def read(path):
    p = Path(path)
    if not p.is_absolute(): p = R/p
    for q in reversed((p,) + tuple(p.parents)):
        s = q.lstat()
        assert not stat.S_ISLNK(s.st_mode), str(q)
        assert stat.S_ISDIR(s.st_mode) if q != p else stat.S_ISREG(s.st_mode) and s.st_nlink == 1, str(q)
    return p.read_bytes()

def pin(path):
    raw = read(path)
    return {'path': str(path), 'size': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}

def load(path): return json.loads(read(path))
def write(name, obj):
    p = D/name
    raw = (json.dumps(obj,ensure_ascii=False,indent=2)+'\n').encode()
    with p.open('xb') as f: f.write(raw)
    assert read(p) == raw
    return pin(str(p.relative_to(R)))
